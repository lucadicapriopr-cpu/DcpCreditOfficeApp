// scripts/importInsoluti.cjs
const fs = require("fs");
const path = require("path");
const XLSX = require("xlsx");
const { PrismaClient } = require("@prisma/client");

const prisma = new PrismaClient();

// 🔹 Parser formato GGMMAAAA o seriale Excel → Date
function parseDateGGMMYYYY(val) {
  if (!val) return null;

  const str = String(val).trim();

  // Caso 1: formato testuale GGMMAAAA (es. "01022025")
  if (/^\d{8}$/.test(str)) {
    const dd = parseInt(str.slice(0, 2), 10);
    const mm = parseInt(str.slice(2, 4), 10) - 1;
    const yyyy = parseInt(str.slice(4, 8), 10);
    return new Date(Date.UTC(yyyy, mm, dd));
  }

  // Caso 2: seriale Excel numerico (es. 45870 → 2025-08-06)
  if (!isNaN(val)) {
    const excelEpoch = new Date(Date.UTC(1899, 11, 30)); // base Excel
    return new Date(excelEpoch.getTime() + (val * 24 * 60 * 60 * 1000));
  }

  console.warn(`⚠️ Data non riconosciuta: "${val}"`);
  return null;
}

async function importInsoluti(filePath) {
  console.log(`📂 Importazione file: ${filePath}`);

  const workbook = XLSX.readFile(filePath);
  const sheetName = workbook.SheetNames[0];
  const sheet = workbook.Sheets[sheetName];
  const rows = XLSX.utils.sheet_to_json(sheet);

  for (const row of rows) {
    try {
      // 🔹 Cliente
      const cliente = await prisma.cliente.upsert({
        where: {
          codiceFiscale:
            String(
              row["Codice Fiscale"] || row["Partita IVA"] || row.Nominativo
            ),
        },
        update: {},
        create: {
          nominativo: row.Nominativo,
          codiceFiscale: row["Codice Fiscale"] || null,
          partitaIva: row["Partita IVA"] || null,
          telefono: row.RecapitoTelefonico
            ? String(row.RecapitoTelefonico)
            : null,
        },
      });

      // 🔹 Utenza
      const utenza = await prisma.utenza.upsert({
        where: {
          clienteId_contratto_podPdr: {
            clienteId: cliente.idCliente,
            contratto: String(row.fa_nmcon),
            podPdr: String(row["Codice POD"] || row["Codice PDR"] || ""),
          },
        },
        update: {},
        create: {
          clienteId: cliente.idCliente,
          contratto: String(row.fa_nmcon),
          podPdr: String(row["Codice POD"] || row["Codice PDR"] || ""),
          comune: row.COMUNE || null,
          stato: row.Stato_Utenza || null,
          descrizioneStato: row.DesStato_Utenza || null,
          dataCessazione: parseDateGGMMYYYY(row.Data_Cessazione),
        },
      });

      // 🔹 Fattura (upsert con importo + incassato)
      await prisma.fattura.upsert({
        where: {
          utenzaId_numero: {
            utenzaId: utenza.idUtenza,
            numero: String(row.numero_fattura),
          },
        },
        update: {
          importo: parseFloat(row.importo_fattura || 0),
          incassato: parseFloat(row.Incassato || 0),
          stato: row.stato || "INSOLUTA",
        },
        create: {
          numero: String(row.numero_fattura),
          dataFattura: parseDateGGMMYYYY(row.DataFattura),
          periodo: row["Periodo Fatturazione"] || null,
          importo: parseFloat(row.importo_fattura || 0),
          scadenza: parseDateGGMMYYYY(row.scadenza),
          stato: row.stato || "INSOLUTA",
          incassato: parseFloat(row.Incassato || 0),
          dataPrimoSollecito: parseDateGGMMYYYY(
            row["Data primo sollecito ancora aperto"]
          ),
          importoPrimoSollecito: parseFloat(
            row["Importo data primo sollecito ancora"] || 0
          ),
          pianoRientro: row["Piano di rientro o rateizzazione"] || null,
          praticaPod: row["Pratica POD"] || null,
          dataPraticaPod: parseDateGGMMYYYY(row["Data pratica POD"]),
          agente: row.Agente || null,
          esclusione: row["EsclusioneContrattoDaElab."] === "SI",
          utenzaId: utenza.idUtenza,
        },
      });

      console.log(
        `✅ Importata/aggiornata fattura ${row.numero_fattura} cliente ${row.Nominativo}`
      );
    } catch (err) {
      console.error(`❌ Errore su riga: ${JSON.stringify(row)}\n`, err);
    }
  }

  await prisma.$disconnect();
}

const filePath = path.join(__dirname, "insoluti.xlsx");
importInsoluti(filePath);
